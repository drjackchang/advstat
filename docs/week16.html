library(tidyverse)
library(gt)

# 生成期末考試資料集
set.seed(1216)
n <- 250

ems_final <- data.frame(
  id           = sprintf("F%03d", 1:n),
  gender       = sample(c("男","女"), n,
                        replace = TRUE,
                        prob    = c(0.70, 0.30)),
  age          = round(rnorm(n, 35, 7)),
  level        = sample(c("EMT-1","EMT-2","EMT-P"), n,
                        replace = TRUE,
                        prob    = c(0.45, 0.35, 0.20)),
  years        = round(runif(n, 1, 25)),
  district     = sample(c("北部","中部","南部"), n,
                        replace = TRUE,
                        prob    = c(0.42, 0.30, 0.28)),
  monthly_cases = round(rnorm(n, 46, 11)),
  night_shifts  = round(rnorm(n, 14, 5)),
  overtime_hrs  = round(rnorm(n, 11, 6)),
  bmi           = round(rnorm(n, 24.3, 3.1), 1),
  sbp           = round(rnorm(n, 117, 13)),
  sleep_hrs     = round(rnorm(n, 6.1, 1.2), 1),
  sleep_qual    = round(rnorm(n, 5.9, 1.9)),
  stress_vas    = round(rnorm(n, 5.5, 1.7), 1),
  mbi_ee        = round(rnorm(n, 3.3, 1.1), 1),
  mbi_dp        = round(rnorm(n, 2.5, 1.0), 1),
  mbi_pa        = round(rnorm(n, 4.0, 0.9), 1),
  job_sat       = round(rnorm(n, 3.2, 0.9)),
  mist_pre      = round(rnorm(n, 61, 10)),
  mist_post     = NA,
  supervisor    = round(rnorm(n, 73, 12)),
  response_time = round(rnorm(n, 9.5, 2.6), 1),
  case_severity = sample(1:4, n, replace = TRUE,
                         prob = c(0.34,0.38,0.19,0.09)),
  occ_injury    = sample(0:1, n, replace = TRUE,
                         prob = c(0.77, 0.23)),
  burnout_risk  = NA
)

# 修正範圍
ems_final$age           <- pmax(22, pmin(58, ems_final$age))
ems_final$monthly_cases <- pmax(18, pmin(82, ems_final$monthly_cases))
ems_final$night_shifts  <- pmax(0,  pmin(30, ems_final$night_shifts))
ems_final$overtime_hrs  <- pmax(0,  pmin(45, ems_final$overtime_hrs))
ems_final$bmi           <- pmax(17, pmin(39, ems_final$bmi))
ems_final$sbp           <- pmax(88, pmin(162, ems_final$sbp))
ems_final$sleep_hrs     <- pmax(3,  pmin(9,  ems_final$sleep_hrs))
ems_final$sleep_qual    <- pmax(0,  pmin(21, ems_final$sleep_qual))
ems_final$stress_vas    <- pmax(0,  pmin(10, ems_final$stress_vas))
ems_final$mbi_ee        <- pmax(0,  pmin(6,  ems_final$mbi_ee))
ems_final$mbi_dp        <- pmax(0,  pmin(6,  ems_final$mbi_dp))
ems_final$mbi_pa        <- pmax(0,  pmin(6,  ems_final$mbi_pa))
ems_final$job_sat       <- pmax(1,  pmin(5,  ems_final$job_sat))
ems_final$mist_pre      <- pmax(30, pmin(95, ems_final$mist_pre))
ems_final$supervisor    <- pmax(40, pmin(100, ems_final$supervisor))
ems_final$response_time <- pmax(2,  pmin(22, ems_final$response_time))

# 訓練後評分（前測 + 進步）
ems_final$mist_post <- pmin(100,
  ems_final$mist_pre + round(rnorm(n, 6.5, 5)))

# 倦怠風險（二元）
ems_final$burnout_risk <- as.integer(
  ems_final$mbi_ee > 3.5 | ems_final$stress_vas > 7)

# 轉為 factor
ems_final$gender   <- factor(ems_final$gender,
                              levels = c("男","女"))
ems_final$level    <- factor(ems_final$level,
                              levels = c("EMT-1","EMT-2","EMT-P"))
ems_final$district <- factor(ems_final$district,
                              levels = c("北部","中部","南部"))

# 儲存
write.csv(ems_final, "data/ems_final.csv",
          row.names = FALSE)
cat("ems_final.csv 已生成，n =", n,
    "，欄位 =", ncol(ems_final), "\n")

題目

第一題（15%）：描述統計

使用 ems_final.csv，完成以下任務：

(a) 製作受試者基本特性表（Table 1），包含：

  • 類別變項：性別、職級、地區（次數與百分比）
  • 連續變項：年齡、年資、BMI、收縮壓、睡眠時數、主觀壓力、情感耗竭、反應時間（Mean ± SD 或 Median（IQR），依分布決定）

製作成 APA 格式 gt 表格。

(b) 計算各職級的職業傷害（occ_injury）盛行率， 計算 EMT-P 相對於 EMT-1 的相對危險性（RR）， 並說明意義。

第二題（25%）：推論統計（比較）

(a) 獨立 t 檢定

比較男性與女性 EMS 人員的平均到院反應時間（response_time）。

要求：說明方法選擇理由 → 假設檢查（Shapiro-Wilk）→ 執行 Welch t 檢定 → 計算 Cohen’s d → gt APA 格式結果表 → APA 報告句 → 實務詮釋。

(b) 單因子 ANOVA

比較三個地區（district）的主觀壓力（stress_vas）。

要求：假設檢查（Shapiro-Wilk + Levene’s test）→ 執行 ANOVA → gt APA 格式摘要表 → Tukey HSD 事後比較表 → APA 報告句。

(c) 成對 t 檢定

比較訓練前（mist_pre）與訓練後（mist_post）的 MIST 評分。

要求：說明為何使用成對而非獨立 t → 假設檢查 → 執行成對 t → 計算 Cohen’s d → 前後測連線圖 → APA 報告句。

第三題（20%）：相關與線性回歸

(a) 相關分析

分析主觀壓力（stress_vas）與情感耗竭（mbi_ee）的關係。

要求：先畫散點圖 → Shapiro-Wilk 常態性檢定 → 選擇 Pearson 或 Spearman（說明理由）→ gt APA 格式結果表 → APA 報告句。

(b) 多元線性回歸

以 stress_vas、night_shifts、sleep_hrs、years 預測情感耗竭（mbi_ee）。

要求：建立模型 → 執行 VIF 診斷 → gt APA 格式係數表（含標準化係數 β*）→ 說明哪個變項預測力最強 → APA 報告句。

第四題（25%）：羅吉斯回歸

以 stress_vas、night_shifts、sleep_hrs、level 預測職業傷害（occ_injury）。

要求：

  1. 製作各組職業傷害盛行率長條圖
  2. 執行羅吉斯回歸
  3. gt APA 格式係數表（含 OR 與 95% CI）
  4. 計算 Nagelkerke R² 與 Hosmer-Lemeshow 適配度
  5. 畫 ROC 曲線並計算 AUC
  6. 完整 APA 報告段落（3–5 句）

第五題（15%）：進階方法（三選一）

選擇以下其中一題作答：

選項 A：中介分析

檢驗夜班次數（night_shifts）對工作滿意度（job_sat） 的影響是否透過主觀壓力（stress_vas）中介。

要求：Baron-Kenny 四步驟表格 + Bootstrap 間接效果表格 + APA 報告句 + 判斷完全或部分中介。

選項 B：重複測量分析

使用 mist_pre 與 mist_post 建立縱貫資料， 以 GEE 分析訓練效果，加入 level 作為共變項。

要求：長格式資料整理 + GEE 執行 + gt APA 格式係數表 + APA 報告句。

選項 C：MANOVA

以 district（三地區）為自變項， 以 stress_vas、mbi_ee、job_sat 為依變項， 執行 MANOVA。

要求：依變項相關矩陣 + Pillai’s Trace 多變量摘要表 + 後續單變量 ANOVA 表（Bonferroni 校正）+ APA 報告段落。

註釋考試提示

遇到困難時的處理順序：

  1. 先確認 X-Y 類型 → 查第 15 週的決策表
  2. 假設檢查出問題 → 查對應週次的假設檢查流程
  3. 語法忘記 → 查課程網站對應週次的 R 實作區塊
  4. 結果詮釋不確定 → 查對應週次的 APA 報告範例

最常見的失分點：

  • p 值詮釋錯誤（把 p < .05 說成「效果很大」）
  • 忘記計算效果量
  • OR 詮釋錯誤（說成機率而非勝算）
  • 信賴區間詮釋錯誤
  • APA 格式不完整（少了 df、CI 或效果量）
